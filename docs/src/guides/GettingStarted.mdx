# 快速开始

> 5 分钟快速上手 Eos Components

## 📦 安装

### 使用包管理器安装

```bash
# 使用 npm
npm install @eosjs/components

# 使用 pnpm (推荐)
pnpm add @eosjs/components

# 使用 yarn
yarn add @eosjs/components
```

### CDN 引入

```html
<!-- 开发版本 -->
<script type="module" src="https://unpkg.com/@eosjs/components/dist/index.js"></script>

<!-- 生产版本 -->
<script type="module" src="https://unpkg.com/@eosjs/components/dist/index.min.js"></script>
```

## 🚀 基础使用

### 1. 注册组件

在使用组件之前，需要先注册 Web Components：

```javascript
import { registerComponents } from '@eosjs/components';

// 注册所有组件
registerComponents();
```

### 2. 在 HTML 中使用

```html
<!DOCTYPE html>
<html lang="zh-CN">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Eos Components Demo</title>
</head>
<body>
  <!-- 按钮组件 -->
  <eos-button type="primary">主要按钮</eos-button>
  <eos-button type="success">成功按钮</eos-button>
  
  <!-- 图片组件 -->
  <eos-image 
    src="https://picsum.photos/300/200" 
    alt="示例图片"
    width="300px"
    height="200px">
  </eos-image>
  
  <!-- 轮播图组件 -->
  <eos-carousel autoplay interval="3000">
    <div style="background: #f0f0f0; padding: 50px; text-align: center;">
      幻灯片 1
    </div>
    <div style="background: #e0e0e0; padding: 50px; text-align: center;">
      幻灯片 2
    </div>
  </eos-carousel>

  <script type="module">
    import { registerComponents } from '@eosjs/components';
    registerComponents();
  </script>
</body>
</html>
```

## ⚛️ 在 React 中使用

### 1. 安装和配置

```bash
# 安装组件库
pnpm add @eosjs/components

# 如果使用 TypeScript，可能需要类型声明
pnpm add -D @types/react
```

### 2. 类型声明

创建 `types.d.ts` 文件：

```typescript
// types.d.ts
declare namespace JSX {
  interface IntrinsicElements {
    'eos-button': React.DetailedHTMLProps<React.HTMLAttributes<HTMLElement>, HTMLElement> & {
      type?: string;
      disabled?: boolean;
      loading?: boolean;
      onEClick?: (event: CustomEvent) => void;
    };
    'eos-image': React.DetailedHTMLProps<React.HTMLAttributes<HTMLElement>, HTMLElement> & {
      src?: string;
      alt?: string;
      width?: string | number;
      height?: string | number;
      circle?: boolean;
      onLoad?: (event: CustomEvent) => void;
      onError?: (event: CustomEvent) => void;
    };
    'eos-carousel': React.DetailedHTMLProps<React.HTMLAttributes<HTMLElement>, HTMLElement> & {
      autoplay?: boolean;
      interval?: string;
      loop?: boolean;
      onChange?: (event: CustomEvent) => void;
    };
  }
}
```

### 3. 在组件中使用

```tsx
// App.tsx
import React, { useEffect } from 'react';
import { registerComponents } from '@eosjs/components';

function App() {
  useEffect(() => {
    // 注册组件
    registerComponents();
  }, []);

  const handleButtonClick = (event: any) => {
    console.log('按钮被点击:', event.detail);
  };

  return (
    <div className="App">
      <h1>Eos Components in React</h1>
      
      {/* 按钮组件 */}
      <eos-button 
        type="primary" 
        onEClick={handleButtonClick}>
        点击我
      </eos-button>
      
      {/* 图片组件 */}
      <eos-image 
        src="https://picsum.photos/300/200"
        alt="React 示例"
        width="300px"
        height="200px"
        circle>
      </eos-image>
      
      {/* 轮播图组件 */}
      <eos-carousel autoplay interval="3000">
        <div style={{ background: '#1976d2', color: 'white', padding: '50px', textAlign: 'center' }}>
          React 幻灯片 1
        </div>
        <div style={{ background: '#388e3c', color: 'white', padding: '50px', textAlign: 'center' }}>
          React 幻灯片 2
        </div>
      </eos-carousel>
    </div>
  );
}

export default App;
```

## 🖖 在 Vue 中使用

### 1. 配置 Vue 项目

```javascript
// main.js
import { createApp } from 'vue';
import { registerComponents } from '@eosjs/components';
import App from './App.vue';

// 注册 Web Components
registerComponents();

const app = createApp(App);
app.mount('#app');
```

### 2. 在组件中使用

```vue
<!-- App.vue -->
<template>
  <div class="app">
    <h1>Eos Components in Vue</h1>
    
    <!-- 按钮组件 -->
    <eos-button 
      type="primary" 
      @e-click="handleButtonClick">
      点击我
    </eos-button>
    
    <!-- 图片组件 -->
    <eos-image 
      src="https://picsum.photos/300/200"
      alt="Vue 示例"
      width="300px"
      height="200px">
    </eos-image>
    
    <!-- 轮播图组件 -->
    <eos-carousel 
      autoplay 
      interval="3000"
      @change="handleSlideChange">
      <div class="slide slide-1">Vue 幻灯片 1</div>
      <div class="slide slide-2">Vue 幻灯片 2</div>
    </eos-carousel>
  </div>
</template>

<script>
export default {
  name: 'App',
  methods: {
    handleButtonClick(event) {
      console.log('按钮被点击:', event.detail);
    },
    handleSlideChange(event) {
      console.log('幻灯片切换:', event.detail);
    }
  }
}
</script>

<style>
.slide {
  padding: 50px;
  text-align: center;
  color: white;
}
.slide-1 { background: #1976d2; }
.slide-2 { background: #388e3c; }
</style>
```

## 🅰️ 在 Angular 中使用

### 1. 配置 Angular 项目

```typescript
// app.config.ts
import { ApplicationConfig } from '@angular/core';
import { CUSTOM_ELEMENTS_SCHEMA } from '@angular/core';

export const appConfig: ApplicationConfig = {
  providers: [
    // 其他配置...
  ],
  // 添加 CUSTOM_ELEMENTS_SCHEMA
};
```

```typescript
// main.ts
import { bootstrapApplication } from '@angular/platform-browser';
import { registerComponents } from '@eosjs/components';
import { AppComponent } from './app/app.component';
import { appConfig } from './app/app.config';

// 注册 Web Components
registerComponents();

bootstrapApplication(AppComponent, appConfig);
```

### 2. 在组件中使用

```typescript
// app.component.ts
import { Component, CUSTOM_ELEMENTS_SCHEMA } from '@angular/core';

@Component({
  selector: 'app-root',
  standalone: true,
  schemas: [CUSTOM_ELEMENTS_SCHEMA],
  template: `
    <div class="app">
      <h1>Eos Components in Angular</h1>
      
      <!-- 按钮组件 -->
      <eos-button 
        type="primary" 
        (e-click)="handleButtonClick($event)">
        点击我
      </eos-button>
      
      <!-- 图片组件 -->
      <eos-image 
        src="https://picsum.photos/300/200"
        alt="Angular 示例"
        width="300px"
        height="200px">
      </eos-image>
      
      <!-- 轮播图组件 -->
      <eos-carousel 
        autoplay 
        interval="3000"
        (change)="handleSlideChange($event)">
        <div class="slide slide-1">Angular 幻灯片 1</div>
        <div class="slide slide-2">Angular 幻灯片 2</div>
      </eos-carousel>
    </div>
  `,
  styles: [`
    .slide {
      padding: 50px;
      text-align: center;
      color: white;
    }
    .slide-1 { background: #1976d2; }
    .slide-2 { background: #388e3c; }
  `]
})
export class AppComponent {
  handleButtonClick(event: any) {
    console.log('按钮被点击:', event.detail);
  }

  handleSlideChange(event: any) {
    console.log('幻灯片切换:', event.detail);
  }
}
```

## 🎨 主题定制

### CSS 变量

Eos Components 支持通过 CSS 变量进行主题定制：

```css
:root {
  /* 主色调 */
  --eos-primary-color: #1976d2;
  --eos-success-color: #4caf50;
  --eos-warning-color: #ff9800;
  --eos-danger-color: #f44336;
  
  /* 字体 */
  --eos-font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif;
  --eos-font-size-base: 14px;
  
  /* 边距 */
  --eos-border-radius: 4px;
  --eos-spacing-unit: 8px;
  
  /* 阴影 */
  --eos-box-shadow: 0 2px 8px rgba(0, 0, 0, 0.1);
}
```

### 暗色主题

```css
[data-theme="dark"] {
  --eos-bg-color: #1a1a1a;
  --eos-text-color: #ffffff;
  --eos-border-color: #333333;
}
```

## ❓ 常见问题

### Q: 为什么选择 Web Components？

A: Web Components 是浏览器原生标准，具有框架无关、长期稳定、性能优秀等优势。

### Q: 是否支持 SSR？

A: 目前 Web Components 在 SSR 环境下有一些限制，我们正在开发 SSR 友好的解决方案。
